class PreferencesRepository(private val sharedPreferences: SharedPreferences) {

    suspend fun saveValue(key: String, value: String) {
        withContext(Dispatchers.IO) { // Switch to IO thread
            sharedPreferences.edit().putString(key, value).apply()
        }
    }

    suspend fun getValue(key: String): String? {
        return withContext(Dispatchers.IO) { // Switch to IO thread
            sharedPreferences.getString(key, null)
        }
    }
}

class MyViewModel(private val repository: PreferencesRepository) : ViewModel() {

    fun saveToPreferences(key: String, value: String) {
        viewModelScope.launch {
            repository.saveValue(key, value) // No need to specify thread
        }
    }

    fun loadFromPreferences(key: String) {
        viewModelScope.launch {
